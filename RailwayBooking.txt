import java.util.Scanner;

public class Main {
    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        while (sc.hasNext()) {

            if (sc.hasNextInt()) {
                int age = sc.nextInt();

                if (age >= 60) {
                    System.out.println("Senior citizen concession applied (50%)");
                } else if (age < 12) {
                    System.out.println("Child concession applied (25%)");
                } else {
                    System.out.println("No concession");
                }
            } else {
                sc.next(); // Ignore names like Ravi, Priya
            }
        }

        sc.close();
    }
}
